        const rubRate = Number(data.rubToNgn || SERVER_CONFIG.rubToNgn || 18);
        const built = [];
        COUNTRIES.forEach(c => {
          const cData = data.prices?.[c.slug];
          if (!cData) return;
          APPS.forEach(a => {
            const pData = cData[a.slug];
            if (!pData) return;
            // Pick lowest cost available operator on 5sim (or first operator if 0 stock)
            let bestCost = Infinity, totalStock = 0;
            Object.values(pData).forEach(op => {
              if (!op || typeof op !== 'object') return;
              const cnt = Number(op.count || 0), cost = Number(op.cost || 0);
              totalStock += cnt;
              if (cnt > 0 && cost > 0 && cost < bestCost) bestCost = cost;
              else if (bestCost === Infinity && cost > 0) bestCost = cost;
            });
            if (bestCost !== Infinity) {
              built.push({ country:c.name, countrySlug:c.slug, app:a.name, appSlug:a.slug, baseNgn:Math.max(1, Math.ceil(bestCost * rubRate)), priceRub:bestCost, stock:totalStock });
            }
          });
        });
        LIVE_5SIM_LIST = built;
        refreshBuyPage();
        if (showMsg) showToast('Live 5sim prices & stock updated!', true);
      } catch (err) {
        LIVE_5SIM_LIST = [];
        errBox.textContent = `5sim API Error: ${err.message}`;
        errBox.style.display = 'block';
        refreshBuyPage();
      }
    }

    async function confirmBuyNumber() {
      const u = getCurrentUser(); if (!u || !pendingCheckout) return;
      if (u.blocked) { $('coErrorBox').textContent = 'Your account is blocked.'; $('coErrorBox').style.display = 'block'; return; }
      const cMeta = getCountryMeta(pendingCheckout.country), aMeta = getAppMeta(pendingCheckout.app);
      const buyBtn = $('coBuyBtn'), errBox = $('coErrorBox');
      errBox.style.display = 'none';
      buyBtn.disabled = true; buyBtn.textContent = 'Buying from 5sim...';

      try {
        // 1. Call 5sim backend to buy a real number
        const res = await fetch('/api/5sim/buy', {
          method: 'POST', headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ country: cMeta.slug, product: aMeta.slug })
        });
        const data = await res.json();
        if (!res.ok || !data.ok) {
          errBox.textContent = data.error || '5sim could not allocate a number.';
          errBox.style.display = 'block';
          buyBtn.disabled = false; buyBtn.textContent = 'Buy Number Now';
          return;
        }

        // 5. Compute real price from 5sim cost + admin profit margin/promo & verify wallet balance
        const realPriceNgn = computeFinalPrice(data.priceNgn);
        if (u.balance < realPriceNgn) {
          // Cancel on 5sim since wallet cannot cover real price
          await fetch(`/api/5sim/cancel/${encodeURIComponent(data.id)}`, { method: 'POST' }).catch(() => {});
          errBox.textContent = `Insufficient wallet balance for real 5sim price (₦${formatNGN(realPriceNgn)}).`;
          errBox.style.display = 'block';
          buyBtn.style.display = 'none'; $('coFundBtn').style.display = 'inline-flex';
          return;
        }

        // Deduct real price from user wallet balance
        u.balance -= realPriceNgn;
        const now = Date.now();
        const expiresMs = data.expires ? new Date(data.expires).getTime() : (now + 20 * 60 * 1000);
        u.transactions.unshift({ id:'TX-'+now, date:new Date(now).toLocaleString(), type:'Buy', description:`5sim: ${pendingCheckout.app} (${pendingCheckout.country})`, method:'Wallet', amount:`-₦${formatNGN(realPriceNgn)}`, balanceAfter:u.balance, status:'Completed' });
        u.orders.unshift({ id:'ORD-'+data.id, providerOrderId:data.id, date:new Date(now).toLocaleString(), createdAt:now, expiresAt:expiresMs, app:pendingCheckout.app, country:pendingCheckout.country, flag:pendingCheckout.flag, number:data.phone, priceNum:realPriceNgn, status:'Waiting for SMS', smsCode:null, smsReceivedAt:null, rated:false });

        appState.adminUnreadOrders = (appState.adminUnreadOrders || 0) + 1;
        appState.lastOrderAlert = `New Order: ${u.email} bought ${pendingCheckout.app} ${pendingCheckout.country}`;
        saveState(); updateAuthUI(); closeModal('checkoutModal');
        if (isAdminAuthenticated() && appState.adminSettings.beepEnabled) playBeep(880);
        showToast(`Real 5sim number acquired: ${data.phone}`, true);
        navigateTo('my-numbers');
        pollAllActiveOrders();
      } catch (err) {
        errBox.textContent = `Network error: ${err.message}`;
        errBox.style.display = 'block';
        buyBtn.disabled = false; buyBtn.textContent = 'Buy Number Now';
      }
    }

    async function check5simOrder(order, manual = false) {
      if (!order || order.status !== 'Waiting for SMS' || !order.providerOrderId) return false;
      try {
        const res = await fetch(`/api/5sim/check/${encodeURIComponent(order.providerOrderId)}`);
        const data = await res.json();
        if (!res.ok || !data.ok) {
          if (manual) showToast(data.error || 'Error checking 5sim');
          return false;
        }
        if (data.sms && data.sms.length > 0) {
          order.smsCode = data.sms[0].code || data.sms[0].text;
          order.smsReceivedAt = new Date().toLocaleTimeString();
          order.status = 'Completed';
          return true;
        }
        if (data.status === 'TIMEOUT' || data.status === 'CANCELED' || data.status === 'BANNED') {
          order.status = data.status === 'CANCELED' ? 'Cancelled' : 'Expired';
          return true;
        }
        if (manual) showToast(`5sim Status: ${data.status} — waiting for SMS...`);
      } catch (err) {
        if (manual) showToast(`Check error: ${err.message}`);
      }
      return false;
    }

    async function pollAllActiveOrders() {

            if (c) { changed = true; if (cu && u.email === cu.email && o.smsCode) completedForCu = o; }
          }
        }
      }
      if (changed) {
        saveState(); updateAuthUI();
        if (currentView === 'my-numbers') renderMyNumbersPage();
        if (currentView === 'admin-dashboard' && isAdminAuthenticated()) renderAdminDashboard();
        if (completedForCu) handleSmsCodeArrival(completedForCu, true);
      }
    }

    async function cancelAndRefundOrder(orderId) {
      const u = getCurrentUser(), order = u?.orders?.find(o => o.id === orderId);
      if (!order || order.status !== 'Waiting for SMS') return;
      if (order.smsCode) { showToast('Refund Policy: No refund after code received.'); return; }
      try {
        const res = await fetch(`/api/5sim/cancel/${encodeURIComponent(order.providerOrderId)}`, { method: 'POST' });
        const d = await res.json();
        if (!res.ok || !d.ok) { showToast(d.error || 'Could not cancel on 5sim.'); return; }
        order.status = 'Cancelled'; u.balance += order.priceNum;
        u.transactions.unshift({id:'TX-REF-'+Date.now(),date:new Date().toLocaleString(),type:'Refund',description:`5sim Cancelled: ${order.app} (${order.country})`,method:'5sim Cancel',amount:'+₦'+formatNGN(order.priceNum),balanceAfter:u.balance,status:'Completed'});
        saveState(); updateAuthUI(); renderMyNumbersPage();
        showToast(`Cancelled on 5sim & ₦${formatNGN(order.priceNum)} refunded.`, true);
      } catch (err) { showToast(`Cancel error: ${err.message}`); }
    }
